cmake_minimum_required(VERSION 3.16)
project(CoreSim CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if (MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(coresim_core src/isa.cpp src/cpu.cpp src/assembler.cpp src/view.cpp)
add_executable(coresim src/main.cpp)
target_link_libraries(coresim coresim_core)
add_executable(coresim_tests tests/test_main.cpp)
target_link_libraries(coresim_tests coresim_core)
target_compile_definitions(coresim_tests PRIVATE PROGRAMS_DIR="${CMAKE_SOURCE_DIR}/programs")
enable_testing()
add_test(NAME coresim_tests COMMAND coresim_tests)
